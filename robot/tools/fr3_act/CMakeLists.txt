cmake_minimum_required(VERSION 3.19)
project(fr3_act LANGUAGES CXX)
option(FR3_ACT_WITH_ROBOT "Build the libfranka online executor" ON)
find_package(Threads REQUIRED)
find_package(ruckig CONFIG REQUIRED)
set(profile "${CMAKE_CURRENT_SOURCE_DIR}/../../ros2/src/franka_lerobot_recorder/config/act_limits.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${profile}")
file(READ "${profile}" profile_json)
foreach(kind velocity acceleration jerk)
  string(JSON count LENGTH "${profile_json}" ${kind})
  if(NOT count EQUAL 7)
    message(FATAL_ERROR "${kind} must contain seven values")
  endif()
  set(values "")
  foreach(j RANGE 0 6)
    string(JSON value GET "${profile_json}" ${kind} ${j})
    list(APPEND values "${value}")
  endforeach()
  string(JOIN ", " ${kind}_values ${values})
endforeach()
foreach(kind torque force)
  if(kind STREQUAL "torque")
    set(expected 7)
  else()
    set(expected 6)
  endif()
  foreach(phase acceleration nominal)
    foreach(bound lower upper)
      set(key "${bound}_${kind}_${phase}")
      string(JSON count LENGTH "${profile_json}" collision ${key})
      if(NOT count EQUAL expected)
        message(FATAL_ERROR "collision.${key} must contain ${expected} values")
      endif()
      set(values "")
      math(EXPR last "${expected}-1")
      foreach(j RANGE 0 ${last})
        string(JSON value GET "${profile_json}" collision ${key} ${j})
        string(JSON value_type TYPE "${profile_json}" collision ${key} ${j})
        if(NOT value_type STREQUAL "NUMBER")
          message(FATAL_ERROR "collision.${key} must contain numbers")
        endif()
        list(APPEND values "${value}")
      endforeach()
      string(JOIN ", " collision_${key}_values ${values})
    endforeach()
  endforeach()
endforeach()
file(SHA256 "${profile}" profile_sha)
string(SUBSTRING "${profile_sha}" 0 8 profile_short)
math(EXPR profile_id "0x${profile_short}")
configure_file(limits.hpp.in generated/act_limits.hpp @ONLY)
set(build_identity "")
foreach(name ../fr3_replay/trajectory.hpp CMakeLists.txt bridge.cpp build_info.hpp.in collision.hpp limits.hpp.in motion.hpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${name}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${name}" source_sha)
  string(APPEND build_identity "${name}:${source_sha}\n")
endforeach()
string(SHA256 build_source_sha "${build_identity}")
configure_file(build_info.hpp.in generated/act_build_info.hpp @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
enable_testing()
add_executable(fr3_act_check check.cpp)
target_compile_features(fr3_act_check PRIVATE cxx_std_17)
target_link_libraries(fr3_act_check PRIVATE ruckig::ruckig)
add_test(NAME act_motion COMMAND fr3_act_check)
add_executable(fr3_act_sim bridge.cpp)
target_compile_features(fr3_act_sim PRIVATE cxx_std_17)
target_compile_options(fr3_act_sim PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(fr3_act_sim PRIVATE Threads::Threads ruckig::ruckig)
if(FR3_ACT_WITH_ROBOT)
  find_package(Franka CONFIG REQUIRED)
  add_executable(fr3_act_bridge bridge.cpp)
  target_compile_features(fr3_act_bridge PRIVATE cxx_std_17)
  target_compile_definitions(fr3_act_bridge PRIVATE FR3_ACT_ROBOT)
  target_compile_options(fr3_act_bridge PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(fr3_act_bridge PRIVATE Franka::Franka Threads::Threads ruckig::ruckig)
endif()
